let atual = 0;

let pontos = 0;

let respondeu = false;

function iniciar() {

  atual = 0;

  pontos = 0;

  respondeu = false;

  document.getElementById("jogo").classList.remove("oculto");

  document.getElementById("resultado").classList.add("oculto");

  mostrarQuestao();

}

function mostrarQuestao() {

  respondeu = false;

  const q = questoes[atual];

  document.getElementById("numero").textContent =

    "Questão " + (atual + 1) + " de " + questoes.length;

  document.getElementById("barra").style.width =

    (atual / questoes.length * 100) + "%";

  document.getElementById("pergunta").textContent = q.p;

  document.getElementById("feedback").style.display = "none";

  document.getElementById("proximo").classList.add("oculto");

  const area = document.getElementById("opcoes");

  area.innerHTML = "";

  q.o.forEach((opcao, i) => {

    const botao = document.createElement("button");

    botao.className = "opcao";

    botao.textContent = String.fromCharCode(65 + i) + ") " + opcao;

    botao.onclick = () => verificar(i);

    area.appendChild(botao);

  });

}

function verificar(escolha) {

  if (respondeu) return;

  respondeu = true;

  const q = questoes[atual];

  const botoes = document.querySelectorAll(".opcao");

  botoes.forEach((botao, i) => {

    botao.disabled = true;

    if (i === q.r) botao.classList.add("correta");

    if (i === escolha && i !== q.r) botao.classList.add("errada");

  });

  const feedback = document.getElementById("feedback");

  feedback.style.display = "block";

  if (escolha === q.r) {

    pontos++;

    feedback.style.background = "#dcfce7";

    feedback.innerHTML = "✅ <strong>Muito bem! Resposta correta!</strong><br>" + q.e;

  } else {

    feedback.style.background = "#fee2e2";

    feedback.innerHTML = "💡 <strong>Quase lá! Vamos aprender:</strong><br>" + q.e;

  }

  document.getElementById("proximo").classList.remove("oculto");

  document.getElementById("proximo").textContent =

    atual === questoes.length - 1 ? "Ver resultado 🏆" : "Próxima questão ➜";

}

function avancar() {

  atual++;

  if (atual < questoes.length) {

    mostrarQuestao();

  } else {

    finalizar();

  }

}

function finalizar() {

  document.getElementById("jogo").classList.add("oculto");

  document.getElementById("resultado").classList.remove("oculto");

  document.getElementById("barra").style.width = "100%";

  document.getElementById("pontuacao").innerHTML =

    "Você acertou <strong>" + pontos + " de " + questoes.length + "</strong> questões!";

  let mensagem;

  if (pontos === 10) mensagem = "🏆 Excelente! Você é fera na formação de palavras!";

  else if (pontos >= 7) mensagem = "👏 Muito bem! Você está dominando o conteúdo!";

  else if (pontos >= 4) mensagem = "👍 Bom trabalho! Continue praticando!";

  else mensagem = "📚 Não desanime! Revise o conteúdo e tente novamente!";

  document.getElementById("mensagem").textContent = mensagem;

}
